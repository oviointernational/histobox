## 1. What This App Is

Histobox is a histopathology laboratory management system. It manages the full workflow of a histology lab:
case entry → bench processing (fixation, processing, embedding, microtomy, staining) → microscopy/QC → sign-out → reporting → reagent management.

The app is a **pre-compiled React bundle** (no source code — only built output). The Replit project serves it via Express.

### Icon Generation (pngjs — run once if icons are missing)
```js
// generate-icons.js
const { PNG } = require('pngjs');
const fs = require('fs');

function createIcon(size, cornerRadius) {
  const png = new PNG({ width: size, height: size, colorType: 6 });
  const r = 74, g = 158, b = 142; // teal #4A9E8E
  for (let y = 0; y < size; y++) {
    for (let x = 0; x < size; x++) {
      const idx = (size * y + x) * 4;
      const cr = cornerRadius;
      const dx = Math.max(cr - x, 0, x - (size - 1 - cr));
      const dy = Math.max(cr - y, 0, y - (size - 1 - cr));
      if (dx * dx + dy * dy <= cr * cr) {
        png.data[idx] = r; png.data[idx+1] = g; png.data[idx+2] = b; png.data[idx+3] = 255;
      } else {
        png.data[idx] = png.data[idx+1] = png.data[idx+2] = 0; png.data[idx+3] = 0;
      }
    }
  }
  const cx = Math.round(size/2), cy = Math.round(size/2);
  const barW = Math.round(size*0.12), halfH = Math.round(size*0.28), halfW = Math.round(size*0.18);
  for (let y = cy-halfH; y <= cy+halfH; y++) {
    for (let x = cx-halfW-barW; x <= cx-halfW; x++) { const i=(size*y+x)*4; png.data[i]=png.data[i+1]=png.data[i+2]=255; png.data[i+3]=255; }
    for (let x = cx+halfW; x <= cx+halfW+barW; x++) { const i=(size*y+x)*4; png.data[i]=png.data[i+1]=png.data[i+2]=255; png.data[i+3]=255; }
  }
  for (let y = cy-Math.round(barW/2); y <= cy+Math.round(barW/2); y++) {
    for (let x = cx-halfW-barW; x <= cx+halfW+barW; x++) { const i=(size*y+x)*4; png.data[i]=png.data[i+1]=png.data[i+2]=255; png.data[i+3]=255; }
  }
  return PNG.sync.write(png);
}
fs.mkdirSync('icons', { recursive: true });
fs.writeFileSync('icons/icon-192.png', createIcon(192, 38));
fs.writeFileSync('icons/icon-512.png', createIcon(512, 102));
console.log('Icons generated.');
```

const fs = require('fs');

function patch(label, from, to, expectCount) {
  const count = code.split(from).length - 1;
  if (count === 0) { console.error('MISS  [' + label + ']'); return; }
  if (expectCount && count !== expectCount) { console.warn('WARN  [' + label + '] expected ' + expectCount + ' got ' + count); }
  code = code.split(from).join(to);
  console.log('OK    [' + label + '] replaced ' + count + 'x');
